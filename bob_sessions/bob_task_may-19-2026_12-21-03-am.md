703 | def _parse_ai_json(text: str) -> Dict[str, Any]:
704 |     """
705 |     Extract and parse a JSON object from a model response.
706 |     Handles: markdown fences, leading explanation, trailing text, trailing commas.
707 |     """
708 |     import json as _json, re as _re  # noqa: PLC0415
709 | 
710 |     # 1. Strip all markdown code fences (```json ... ``` or ``` ... ```)
711 |     text = _re.sub(r'```(?:json)?\s*', '', text)
712 |     text = _re.sub(r'```', '', text).strip()

714 |     # 2. Try direct parse (model obeyed instructions perfectly)
715 |     try:
716 |         return _json.loads(text)
717 |     except _json.JSONDecodeError:

720 |     # 3. Brace-counting extraction of first complete {...} block
721 |     extracted = _brace_extract(text)

724 |         try:
725 |             return _json.loads(extracted)

728 |         # 3b. Repair trailing commas (common model mistake)
729 |         repaired = _re.sub(r',\s*([}\]])', r'\1', extracted)
730 |         try:

732 |         except _json.JSONDecodeError:

741 | def _parse_ai_array(text: str) -> List[Dict[str, Any]]:
742 |     """
743 |     Sibling of _parse_ai_json for responses that should be a JSON array.
744 |     Granite likes to wrap arrays in "[Answer]" prefixes - take the last [...].
745 |     """
746 |     import json as _json, re as _re  # noqa: PLC0415
747 | 
748 |     text = _re.sub(r'```(?:json)?\s*', '', text)
749 |     text = _re.sub(r'```', '', text).strip()
750 | 
751 |     try:
752 |         result = _json.loads(text)
753 |         if isinstance(result, list):
754 |             return result
755 |     except _json.JSONDecodeError:
756 |         pass
757 | 
758 |     # Find the LAST balanced [...] block. Greedy from the back so we skip
759 |     # any "[Answer]"-style preludes.
760 |     end = text.rfind(']')
761 |     if end == -1:
762 |         return []
763 |     depth = 0
764 |     start = -1
765 |     for i in range(end, -1, -1):
766 |         ch = text[i]
767 |         if ch == ']':
768 |             depth += 1
769 |         elif ch == '[':
770 |             depth -= 1
771 |             if depth == 0:
772 |                 start = i
773 |                 break
774 |     if start == -1:
775 |         return []
776 | 
777 |     candidate = text[start : end + 1]
778 |     try:
779 |         result = _json.loads(candidate)
780 |         return result if isinstance(result, list) else []
781 |     except _json.JSONDecodeError:
782 |         pass
783 | 
784 |     # Repair trailing commas
785 |     repaired = _re.sub(r',\s*([}\]])', r'\1', candidate)
786 |     try:
787 |         result = _json.loads(repaired)
788 |         return result if isinstance(result, list) else []
789 |     except _json.JSONDecodeError:
790 |         return []

793 | def _brace_extract(text: str) -> Optional[str]:
794 |     """Return the first syntactically complete {...} JSON object in text."""
795 |     start = text.find('{')
796 |     if start == -1:
797 |         return None
798 |     depth = 0
799 |     in_string = False
800 |     escape = False
801 |     for i, ch in enumerate(text[start:], start):
802 |         if escape:
803 |             escape = False
804 |             continue
805 |         if ch == '\\' and in_string:
806 |             escape = True
807 |             continue
808 |         if ch == '"':
809 |             in_string = not in_string
810 |             continue
811 |         if in_string:
812 |             continue
813 |         if ch == '{':
814 |             depth += 1
815 |         elif ch == '}':
816 |             depth -= 1
817 |             if depth == 0:
818 |                 return text[start : i + 1]
819 |     return None

158 | def detect_main_file(workspace: Path) -> Optional[Path]:
159 |     """
160 |     Detect the main entry point file in a workspace.
161 |     
162 |     Detection strategy:
163 |     1. Check for common entry point filenames (main.py, app.js, etc.)
164 |     2. Check in common subdirectories (src/, app/, backend/, etc.)
165 |     3. Scan for API framework indicators (FastAPI, Express, Spring, etc.)
166 |     
167 |     Args:
168 |         workspace: Workspace root directory
169 |         
170 |     Returns:
171 |         Path to main entry point file, or None if not found
172 |     """
173 |     # Check root directory for common entry point names
174 |     for name in _ENTRY_NAMES:
175 |         c = workspace / name
176 |         if c.exists() and c.is_file():
177 |             return c
178 |     
179 |     # Check common subdirectories
180 |     for subdir in _ENTRY_SUBDIRS:
181 |         sub = workspace / subdir
182 |         if not sub.is_dir():
183 |             continue
184 |         for name in _ENTRY_NAMES:
185 |             c = sub / name
186 |             if c.exists() and c.is_file():
187 |                 return c
188 |     
189 |     # Scan for API framework indicators
190 |     for fp in sorted(workspace.rglob("*")):
191 |         if not fp.is_file() or should_skip(fp):
192 |             continue
193 |         lang = detect_language(fp)
194 |         if not lang:
195 |             continue
196 |         indicators = _API_INDICATORS.get(lang, [])
197 |         if not indicators:
198 |             continue
199 |         try:
200 |             content = fp.read_text(encoding="utf-8", errors="ignore")
201 |         except OSError:
202 |             continue
203 |         if any(ind in content for ind in indicators):
204 |             return fp
205 |     
206 |     return None

274 | def parse_python_file(file_path: Path) -> List[Dict[str, Any]]:
275 |     """
276 |     Parse Python file using AST for maximum accuracy.
277 |     
278 |     Extracts:
279 |     - Top-level function definitions (sync and async)
280 |     - FastAPI/Flask route decorators (@app.get, @app.post, etc.)
281 |     - Function call chains within each function
282 |     - Line numbers and code snippets

284 |     Args:
285 |         file_path: Path to Python source file
286 |         
287 |     Returns:
288 |         List of node dictionaries with keys:
289 |         - name: Function name
290 |         - type: 'endpoint' or 'function'
291 |         - method: HTTP method (for endpoints)
292 |         - route: Route path (for endpoints)
293 |         - code: Code snippet
294 |         - line: Line number
295 |         - calls: List of function names called
296 |     """
297 |     try:
298 |         source = file_path.read_text(encoding="utf-8")
299 |         tree = ast.parse(source, filename=str(file_path))
300 |     except (SyntaxError, OSError):
301 |         return []
302 | 
303 |     lines = source.splitlines()
304 | 
305 |     class _CC(ast.NodeVisitor):
306 |         """Call collector - extracts function call names from AST."""
307 |         def __init__(self) -> None:
308 |             self.calls: List[str] = []
309 |             self._seen: Set[str] = set()
310 | 
311 |         def visit_Call(self, node: ast.Call) -> None:  # type: ignore[override]
312 |             name = ""
313 |             if isinstance(node.func, ast.Name):
314 |                 name = node.func.id
315 |             elif isinstance(node.func, ast.Attribute):
316 |                 name = node.func.attr
317 |             if name and name not in self._seen:
318 |                 self._seen.add(name)
319 |                 self.calls.append(name)
320 |             self.generic_visit(node)
321 | 
322 |     nodes: List[Dict[str, Any]] = []
323 | 
324 |     # Parse only top-level functions (not nested or class methods)
325 |     for node in tree.body:
326 |         if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
327 |             continue
328 | 
329 |         # Extract code snippet
330 |         start = max(getattr(node, "lineno", 1) - 1, 0)
331 |         end = getattr(node, "end_lineno", start + 15)
332 |         code = "\n".join(lines[start:end]).rstrip()
333 | 
334 |         # Collect function calls
335 |         cc = _CC()
336 |         cc.visit(node)
337 | 
338 |         # Detect HTTP endpoint decorators
339 |         method: Optional[str] = None
340 |         route: Optional[str] = None
341 |         node_type = "function"
342 | 
343 |         for dec in node.decorator_list:
344 |             if isinstance(dec, ast.Call) and isinstance(dec.func, ast.Attribute):
345 |                 attr = dec.func.attr.lower()
346 |                 if attr in _HTTP_METHODS:
347 |                     route_path = "/"
348 |                     if dec.args and isinstance(dec.args[0], ast.Constant):
349 |                         route_path = str(dec.args[0].value)
350 |                     method = attr.upper()
351 |                     route = route_path
352 |                     node_type = "endpoint"
353 |                     break
354 |             elif isinstance(dec, ast.Attribute) and dec.attr.lower() in _HTTP_METHODS:
355 |                 method = dec.attr.upper()
356 |                 route = "/"
357 |                 node_type = "endpoint"
358 |                 break
359 | 
360 |         nodes.append({
361 |             "name": node.name,
362 |             "type": node_type,
363 |             "method": method,
364 |             "route": route,
365 |             "code": code,
366 |             "line": getattr(node, "lineno", 1),
367 |             "calls": cc.calls,
368 |         })
369 | 
370 |     return nodes

def safe_relative(file_path: Path, root: Path) -> str:
    """
    Get relative path from root, falling back to absolute if not relative.
    
    Args:
        file_path: File path to make relative
        root: Root directory
        
    Returns:
        Relative path string with forward slashes
    """
    try:
        return str(file_path.resolve().relative_to(root.resolve())).replace("\\", "/")
    except ValueError:
        return str(file_path.resolve()).replace("\\", "/")


def is_within(path: Path, root: Path) -> bool:
    """
    Check if path is within root directory.
    
    Args:
        path: Path to check
        root: Root directory
        
    Returns:
        True if path is within root
    """
    try:
        path.resolve().relative_to(root.resolve())
        return True
    except ValueError:
        return False


def validate_workspace_path(path_value: str) -> Path:
    """
    Validate and resolve workspace directory path.
    
    Args:
        path_value: Path string to validate
        
    Returns:
        Resolved Path object
        
    Raises:
        HTTPException: If path is invalid
    """
    workspace = Path(path_value).resolve()
    if not workspace.exists():
        raise HTTPException(status_code=400, detail="Local directory path does not exist!")
    if not workspace.is_dir():
        raise HTTPException(status_code=400, detail="Provided workspace path is not a directory!")
    return workspace


def validate_file_path(path_value: str, must_exist: bool = True) -> Path:
    """
    Validate and resolve file path.
    
    Args:
        path_value: Path string to validate
        must_exist: Whether file must exist
        
    Returns:
        Resolved Path object
        
    Raises:
        HTTPException: If path is invalid
    """
    raw = (path_value or "").strip()
    if not raw:
        raise HTTPException(status_code=400, detail="No path provided.")
    
    file_path = Path(raw).expanduser().resolve()
    
    if must_exist:
        if not file_path.exists():
            raise HTTPException(
                status_code=400,
                detail=f"File path does not exist: {file_path}",
            )
        if file_path.is_dir():
            raise HTTPException(
                status_code=400,
                detail=(
                    f"Path is a directory, not a file: {file_path}. "
                    "Point at the project's main file."
                ),
            )
        if not file_path.is_file():
            raise HTTPException(
                status_code=400,
                detail=f"Path is not a regular file: {file_path}",
            )
    
    return file_path

    """
    Check if string is a GitHub URL.
    
    Args:
        value: String to check
        
    Returns:
        True if value is a GitHub URL

    value = value.strip().lower()

    must_exist: bool = True
) -> Path:
    """

    Args:

    Returns:
        Resolved Path object

        resolved = candidate.resolve()
    elif workspace_root:

Import Utilities
================

Utilities for handling dynamic imports with fallback patterns.
"""

import logging
from typing import Any, Callable, Optional, TypeVar

logger = logging.getLogger(__name__)

T = TypeVar('T')


def try_import(
    primary_module: str,
    fallback_module: str,
    *names: str,
    error_message: Optional[str] = None
) -> dict[str, Any]:
    """
    Try importing from primary module, fall back to secondary.
    
    Args:
        primary_module: Primary module path (e.g., "mcp_service")
        fallback_module: Fallback module path (e.g., "backend.mcp_service")
        *names: Names to import from module
        error_message: Optional error message if both imports fail
        
    Returns:
        Dictionary mapping names to imported objects
        
    Raises:
        ImportError: If both imports fail and error_message is provided
    """
    result = {}
    
    try:
        module = __import__(primary_module, fromlist=names)
        for name in names:
            result[name] = getattr(module, name)
        return result
    except (ModuleNotFoundError, ImportError):
        pass
    
    try:
        module = __import__(fallback_module, fromlist=names)
        for name in names:
            result[name] = getattr(module, name)
        return result
    except (ModuleNotFoundError, ImportError) as exc:
        if error_message:
            raise ImportError(error_message) from exc
        logger.warning(
            "Failed to import %s from %s or %s",
            names, primary_module, fallback_module
        )
        return {}


def safe_import_module(
    primary_module: str,
    fallback_module: Optional[str] = None,
    on_error: Optional[Callable[[Exception], None]] = None
) -> Optional[Any]:
    """
    Safely import a module with optional fallback.
    
    Args:
        primary_module: Primary module path
        fallback_module: Optional fallback module path
        on_error: Optional callback for handling errors
        
    Returns:
        Imported module or None if import fails
    """
    try:
        return __import__(primary_module)
    except (ModuleNotFoundError, ImportError) as exc:
        if fallback_module:
            try:
                return __import__(fallback_module)
            except (ModuleNotFoundError, ImportError) as fallback_exc:
                if on_error:
                    on_error(fallback_exc)
                return None
        if on_error:
            on_error(exc)
        return None


def import_with_fallback(
    module_name: str,
    package_prefix: Optional[str] = None
) -> Optional[Any]:
    """
    Import module with automatic package prefix fallback.
    
    Args:
        module_name: Module name to import
        package_prefix: Optional package prefix (e.g., "backend")
        
    Returns:
        Imported module or None
    """
    try:
        return __import__(module_name)
    except (ModuleNotFoundError, ImportError):
        if package_prefix:
            try:
                full_name = f"{package_prefix}.{module_name}"
                return __import__(full_name, fromlist=[module_name])
            except (ModuleNotFoundError, ImportError):
                pass
        return None

Code Analysis Utilities
=======================

Utilities for parsing and analyzing Python code.

import ast
from pathlib import Path
from typing import Any, Dict, List, Optional, Set, Tuple


def extract_function_names(source: str) -> List[str]:
    """
    Extract top-level function names from Python source code.
    
    Args:
        source: Python source code
        
    Returns:
        List of function names
    """
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return []
    
    names: List[str] = []
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            names.append(node.name)
    return names


def locate_function_range(source: str, function_name: str) -> Optional[Tuple[int, int]]:
    """
    Locate line range of a function in source code.
    
    Args:
        source: Python source code
        function_name: Name of function to locate
        
    Returns:
        Tuple of (start_line, end_line) or None if not found
    """
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return None
    
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            if node.name == function_name:
                start = getattr(node, "lineno", 1) - 1
                end = getattr(node, "end_lineno", start + 1)
                return (start, end)
    
    return None


def extract_called_names(source: str) -> List[str]:
    """
    Extract function call names from Python source code.
    
    Args:
        source: Python source code
        
    Returns:
        List of called function names
    """
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return []
    
    class CallCollector(ast.NodeVisitor):
        def __init__(self) -> None:
            self.calls: List[str] = []
            self._seen: Set[str] = set()
        
        def visit_Call(self, node: ast.Call) -> None:
            name = ""
            if isinstance(node.func, ast.Name):
                name = node.func.id
            elif isinstance(node.func, ast.Attribute):
                name = node.func.attr
            
            if name and name not in self._seen:
                self._seen.add(name)
                self.calls.append(name)
            
            self.generic_visit(node)
    
    collector = CallCollector()
    collector.visit(tree)
    return collector.calls

    Validate Python source code syntax.

        Error message if invalid, None if valid
    """

    """
    Extract import statements from Python source code.
    
    Args:
        source: Python source code
        
    Returns:
        List of imported module names

    Args:

from typing import Optional

def clone_github_repo(github_url: str, target_dir: Optional[Path] = None) -> Path:
    """
    Clone a GitHub repository to a temporary or specified directory.
    
    Args:
        github_url: GitHub repository URL
        target_dir: Optional target directory (creates temp dir if None)
        
    Returns:
        Path to cloned repository
        
    Raises:
        HTTPException: If cloning fails
    """
    if target_dir is None:
        target_dir = Path(tempfile.mkdtemp(prefix="bob-clone-"))
    else:
        target_dir = Path(target_dir)
        target_dir.mkdir(parents=True, exist_ok=True)
    
    logger.info(f"Cloning {github_url} to {target_dir}")
    
    try:
        result = subprocess.run(
            ["git", "clone", "--depth", "1", github_url, str(target_dir)],
            capture_output=True,
            text=True,
            timeout=120,
            check=True,
        )
        logger.info(f"Clone successful: {result.stdout}")
        return target_dir
    except subprocess.CalledProcessError as exc:
        error_msg = f"Git clone failed: {exc.stderr}"
        logger.error(error_msg)
        # Clean up failed clone
        if target_dir.exists():
            shutil.rmtree(target_dir, ignore_errors=True)
        raise HTTPException(status_code=400, detail=error_msg) from exc
    except subprocess.TimeoutExpired as exc:
        error_msg = "Git clone timed out after 120 seconds"
        logger.error(error_msg)
        if target_dir.exists():
            shutil.rmtree(target_dir, ignore_errors=True)
        raise HTTPException(status_code=408, detail=error_msg) from exc
    except FileNotFoundError as exc:
        error_msg = "Git command not found. Please install Git."
        logger.error(error_msg)
        raise HTTPException(status_code=500, detail=error_msg) from exc


def cleanup_clone_dir(clone_dir: Path) -> None:
    """
    Clean up a cloned repository directory.
    
    Args:
        clone_dir: Directory to clean up
    """
    if clone_dir and clone_dir.exists():
        try:
            shutil.rmtree(clone_dir, ignore_errors=True)
            logger.info(f"Cleaned up clone directory: {clone_dir}")
        except Exception as exc:
            logger.warning(f"Failed to clean up {clone_dir}: {exc}")


def is_git_repository(path: Path) -> bool:
    """
    Check if a directory is a Git repository.
    
    Args:
        path: Directory path to check
        
    Returns:
        True if directory is a Git repository
    """

<content>"""
Code Generation Utilities
=========================

Utilities for code generation, formatting, and manipulation.
"""

import ast
from pathlib import Path
from typing import List, Optional


def strip_code_fences(code: str) -> str:
    """
    Remove markdown code fences from generated code.
    
    Args:
        code: Code string potentially wrapped in markdown fences
        
    Returns:
        Code without fences
    """
    text = (code or "").strip()
    if text.startswith("```"):
        text = text.split("\n", 1)[-1]
    if text.endswith("```"):
        text = text.rsplit("```", 1)[0]
    return text.strip()


def ensure_trailing_newline(text: str) -> str:
    """
    Ensure text ends with a newline character.
    
    Args:
        text: Text to check
        
    Returns:
        Text with trailing newline
    """
    return text if text.endswith("\n") else f"{text}\n"


def generate_endpoint_decorator(route_method: str, route_path: str) -> List[str]:
    """
    Generate FastAPI endpoint decorator lines.
    
    Args:
        route_method: HTTP method (GET, POST, etc.)
        route_path: Route path
        
    Returns:
        List of decorator lines
    """
    if not route_method or not route_path:
        return []
    
    method = route_method.strip().upper()
    path = route_path.strip()
    if not method or not path:
        return []
    
    decorator_map = {
        "GET": f'@app.get("{path}")',
        "POST": f'@app.post("{path}")',
        "PUT": f'@app.put("{path}")',
        "DELETE": f'@app.delete("{path}")',
        "PATCH": f'@app.patch("{path}")',
    }
    
    return [decorator_map.get(method, f'@app.api_route("{path}", methods=["{method}"])')]


def inject_decorator(code_block: str, decorator_lines: List[str]) -> str:
    """
    Inject decorator lines before function definition.
    
    Args:
        code_block: Code block containing function
        decorator_lines: Decorator lines to inject
        
    Returns:
        Code block with decorators injected
    """
    if not decorator_lines:
        return code_block
    
    lines = code_block.splitlines()
    
    # Check if decorators already exist
    if any(line.lstrip().startswith("@app.") or line.lstrip().startswith("@router.") for line in lines):
        return code_block
    
    # Find function definition line
    insert_index = None
    for index, line in enumerate(lines):
        stripped = line.lstrip()
        if stripped.startswith("def ") or stripped.startswith("async def "):
            insert_index = index
            break
    
    if insert_index is None:
        return code_block
    
    return "\n".join([*lines[:insert_index], *decorator_lines, *lines[insert_index:]])


def append_code_block(source_code: str, generated_code: str) -> str:
    """
    Append generated code to existing source code with validation.
    
    Args:
        source_code: Existing source code
        generated_code: Generated code to append
